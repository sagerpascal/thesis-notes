% ===========================================================================
% thesis_style.tex
%
% The single home for how the thesis looks. Colours, fonts, headings,
% captions, links, tables, and the furniture every paper backed chapter opens
% with. Nothing here changes what the text says, so it is safe to touch
% without re-reading the style guide in CLAUDE.md.
%
% This file is \input at the very end of the preamble in main.tex, after every
% package has loaded, so anything set here wins.
% ===========================================================================


% ---------------------------------------------------------------------------
% 1. Palette
%
% Five colours carry the whole document. Change them here and everything
% follows. Slate teal for structure, amber for the things a reader clicks or
% looks up, grey for anything that should recede.
% ---------------------------------------------------------------------------

\definecolor{ThesisPrimary}{HTML}{10403B}   % slate teal, headings and rules
\definecolor{ThesisSecondary}{HTML}{276A62} % mid teal, subordinate headings
\definecolor{ThesisAccent}{HTML}{B07A1E}    % amber, citations and URLs
\definecolor{ThesisGrey}{HTML}{5A6672}      % margin notes, page numbers
\definecolor{ThesisRule}{HTML}{C9D8D4}      % hairlines inside boxes
\definecolor{ThesisTint}{HTML}{F1F6F4}      % box backgrounds

% Kept so that older text referring to the previous red palette still
% compiles. Both now point at the current palette.
\colorlet{ThesisDarkRed}{ThesisPrimary}
\colorlet{ThesisAccentRed}{ThesisSecondary}
\colorlet{ThesisGray}{ThesisGrey}


% ---------------------------------------------------------------------------
% 2. Typography
%
% The fonts are kaobook's and are deliberately left alone. Its `fontmethod`
% class option picks them: `tex`, the default, loads Palatino through
% newpxtext and newpxmath, and `modern` loads Libertinus through fontspec.
% `modern` needs XeLaTeX or LuaLaTeX, and this thesis builds with pdflatex, so
% Palatino is what we get. It is a good scientific book face, its maths is
% excellent, and kao.sty tunes \linespread to 1.07 specifically for it.
%
% Do not load a font package here. Loading libertinus or newtx on top of
% newpxtext leaves the text in Palatino while the maths is split between two
% families, which looks fine on a page of prose and wrong on a page of
% equations. Changing the face means changing the class option and the engine.
%
% kaobook loads microtype for us.
% ---------------------------------------------------------------------------

% Let TeX stretch a little further before it reports an overfull line. The
% alternative is hand breaking paragraphs inside imported papers, which we do
% not edit.
\setlength{\emergencystretch}{2em}

% Part pages. KOMA centres the title vertically between \partheadstartvskip
% and \partheadendvskip and nothing else, which leaves a lot of empty paper.
% A hairline above and below gives the title something to sit in.
% \hrule is used rather than \rule because these hooks run in vertical mode.
% \rule would build an hbox at \linewidth, which on a wide part page overruns
% the measure and reports an overfull box on every part.
\renewcommand*{\partheadstartvskip}{%
    \null\vfil
    {\color{ThesisRule}\hrule height 0.4pt}%
    \vspace{1.8\baselineskip}%
}
\renewcommand*{\partheadendvskip}{%
    \vspace{1.8\baselineskip}%
    {\color{ThesisRule}\hrule height 0.4pt}%
    \vfil\newpage
}

    filecolor  = ThesisAccent,

% ---------------------------------------------------------------------------
% 5a. Old two-letter font commands
%
% KOMA turns \rm, \it, \sc and the rest into errors, and the imported papers
% still use them: the Koopman paper writes $\tau_{\rm EMA}$ and the CNA paper
% uses \it in a caption. \DeclareOldFontCommand is the kernel's own way to
% restore them with the right behaviour in text and in maths. It only works in
% a preamble, so unlike the rest of the import compatibility this cannot be
% scoped to the imported chapters. Thesis prose should still use \textit and
% \textsc.
% ---------------------------------------------------------------------------

\DeclareOldFontCommand{\rm}{\normalfont\rmfamily}{\mathrm}
\DeclareOldFontCommand{\sf}{\normalfont\sffamily}{\mathsf}
\DeclareOldFontCommand{\tt}{\normalfont\ttfamily}{\mathtt}
\DeclareOldFontCommand{\bf}{\normalfont\bfseries}{\mathbf}
\DeclareOldFontCommand{\it}{\normalfont\itshape}{\mathit}
\DeclareOldFontCommand{\sl}{\normalfont\slshape}{\@nomath\sl}
\DeclareOldFontCommand{\sc}{\normalfont\scshape}{\@nomath\sc}

% ---------------------------------------------------------------------------
% 5b. Code listings
%
% listings reads its body byte by byte, so a multibyte character inside a
% listing defeats inputenc and the run dies with "Invalid UTF-8 byte sequence".
% One such character appears in the imported papers, an en dash in a COGITAO
% prompt. `literate` is the package's own way of mapping them, and the few
% extra entries here cover the punctuation a future re import is likely to
% bring with it.
% ---------------------------------------------------------------------------

\lstset{
    extendedchars = true,
    literate = {–}{{-}}1 {—}{{---}}1 {‘}{{`}}1 {’}{{'}}1
               {“}{{``}}1 {”}{{''}}1 {…}{{...}}3 {→}{{$\rightarrow$}}1
               {≤}{{$\leq$}}1 {≥}{{$\geq$}}1 {×}{{$\times$}}1 {−}{{-}}1,
}

% ---------------------------------------------------------------------------
% 5c. Float placement
%
% The imported papers were written for two column templates with wide text
% blocks, and they carry a lot of floats: the COGITAO appendix alone has ten
% wide tables in a row. kaobook's text block is narrower, so LaTeX's default
% limits of three floats per page and 70 percent of a page fill the deferral
% queue faster than it drains, and the output routine then spins on the same
% page forever rather than failing. Relaxing the limits lets the queue drain.
% ---------------------------------------------------------------------------

\setcounter{topnumber}{5}
\setcounter{bottomnumber}{3}
\setcounter{totalnumber}{12}
\setcounter{dbltopnumber}{5}
\renewcommand{\topfraction}{0.92}
\renewcommand{\bottomfraction}{0.8}
\renewcommand{\textfraction}{0.05}
\renewcommand{\floatpagefraction}{0.6}
\renewcommand{\dblfloatpagefraction}{0.6}

% A page given over to floats is centred vertically by default, because
% \@fptop and \@fpbot are both infinitely stretchable. One table on such a
% page then sits a third of the way down with white space above and below it
% and reads as an accident. Pinning the top and letting the bottom stretch
% aligns it with the text block on every other page, so a float page looks
% like a decision. \@fpsep keeps some give, otherwise two floats on one page
% are jammed together.
\makeatletter
\setlength{\@fptop}{0pt}
\setlength{\@fpsep}{12pt plus 1fil}
\setlength{\@fpbot}{0pt plus 1fil}
\makeatother

% ---------------------------------------------------------------------------
% 9. Margin citations
%
% kaobiblio's \sidecite sets the usual numeric marker in the text and puts the
% work's author, year and title in the margin, so a reader can see what is
% being cited without turning to the bibliography. kaobook's wide margin is
% there for exactly this.
%
% It is applied by patching the three biblatex commands the thesis actually
% reaches: \cite, \parencite and \textcite. Everything else routes through
% them, because main.tex defines \citet, \citep and \citepeg as wrappers, so
% patching three commands covers all nine papers without touching one of them.
%
% Two rules decide whether a citation goes to the margin.
%
% A citation of several works at once does not. Five full references stacked
% in the margin swamp the page, and the marker in the text is one bracket
% either way. Single key citations are the ones a reader might want to look
% up in place.
%
% A citation outside running text does not. The margin beside a float already
% carries that float's caption, a caption's citation is re-executed when the
% list of figures is typeset, and a table cell has no margin of its own. In
% all of those the margin is either taken or does not exist, so the plain
% citation is used.
%
% And a work already introduced in this chapter does not. Printing the same
% author, year and title five times down one margin tells the reader nothing
% the first one did not: of the 588 single key citations in the thesis, only
% 276 are a work's first mention in its chapter.
%
% Even 276 is more than the column holds, and getting from there to 165 that
% fit took three more decisions. Each was measured, and the measurements are
% the reason to keep them.
%
% The note is the marker, the author and the year, and not the title.
% kaobiblio's own format ends with the full title, which is four or five lines
% in a 136pt column and is the one part the bibliography already carries in
% full.
%
% At most one citation per paragraph reaches the margin. A paragraph that
% cites three works in two sentences otherwise puts three notes in the same
% few lines.
%
% The note is placed by the marginnote package and not by \marginpar. kaobook
% sends an offset-less \marginnote through the floating \marginpar, and that
% was tried first: at full density seventy notes a run were being relocated
% away from the paragraph they belonged to, and one left pending at the
% \clearpage that ends an imported chapter made LaTeX report `Float(s) lost'
% and throw it away. An empty importedpaper block was enough to reproduce
% that, so it is the mechanism and not the volume. Giving the note an offset
% of 0pt routes it through marginnote, which never floats, never relocates and
% cannot be lost. What it does instead is overprint, so the two rules above
% are what make it safe, and the check is to look at the densest margin in the
% document rather than to trust the log.
% ---------------------------------------------------------------------------

\makeatletter

% kaobiblio's own margin format is marker, author, year and the full title,
% which runs to four or five lines in a 136pt column. The title is what makes
% it long and it is the part the bibliography already carries in full, so the
% margin keeps the marker, the author and the year. That is what a reader
% needs to place a citation without turning to the back.
\RenewDocumentCommand{\formatmargincitation}{m}{%
  \parencite{#1}: \citeauthor*{#1}
  (\iffieldundef{year}{\bibsstring{nodate}}{\printfield{year}})%
}

\newif\ifts@sidecite \ts@sidecitetrue   % the feature switch
\newif\ifts@margfree \ts@margfreetrue   % is the margin usable right here?
\newif\ifts@parfree  \ts@parfreetrue    % has this paragraph had one yet?
\newif\ifts@useside

% One per paragraph. The kernel's paragraph hook is the only place that knows
% where a paragraph starts, and it costs nothing when the feature is off.
\AddToHook{para/begin}{\ts@parfreetrue}

% The flag is per chapter, and the chapter number is part of its name rather
% than something to reset, so the appendix and the front matter get their own
% without any bookkeeping.
\newcommand{\ts@markseen}[1]{%
  \expandafter\gdef\csname ts@seen@\thechapter @#1\endcsname{}}

\newcommand{\ts@decide}[1]{%
  \ts@usesidefalse
  \ifts@sidecite
    \ifts@margfree
      \ifts@parfree
        \in@{,}{#1}%
        \ifin@\else
          \@ifundefined{ts@seen@\thechapter @#1}{%
            \ts@usesidetrue \ts@markseen{#1}\ts@parfreefalse
          }{}%
        \fi
      \fi
    \fi
  \fi
}

% \sidecite and its siblings set the in-text marker by calling \cite,
% \parencite and \textcite. Those are the commands being patched, so the
% originals have to be back in scope before one of them is called, or it calls
% itself for ever.
\newcommand{\ts@plaincites}{%
  \let\cite\ts@save@cite
  \let\parencite\ts@save@parencite
  \let\textcite\ts@save@textcite
}

% biblatex takes [prenote][postnote]{keys}; \sidecite takes a vertical shift
% first, so the notes shift one place along. The shift is given as 0pt rather
% than left out, because kaobook reads an absent one as a request for the
% floating \marginpar. See above for why that matters.
\newcommand{\ts@dispatch}[5]{%
  \ts@decide{#5}%
  \ifts@useside
    {\ts@plaincites
     \IfNoValueTF{#3}{#1[0pt]{#5}}%
       {\IfNoValueTF{#4}{#1[0pt][#3]{#5}}{#1[0pt][#3][#4]{#5}}}}%
  \else
    \IfNoValueTF{#3}{#2{#5}}%
      {\IfNoValueTF{#4}{#2[#3]{#5}}{#2[#3][#4]{#5}}}%
  \fi
}

\NewDocumentCommand{\ts@cite}{oom}{%
  \ts@dispatch{\sidecite}{\ts@save@cite}{#1}{#2}{#3}}
\NewDocumentCommand{\ts@parencite}{oom}{%
  \ts@dispatch{\sideparencite}{\ts@save@parencite}{#1}{#2}{#3}}
\NewDocumentCommand{\ts@textcite}{oom}{%
  \ts@dispatch{\sidetextcite}{\ts@save@textcite}{#1}{#2}{#3}}

% Everywhere the margin is not running text. The switch is a plain \newif, so
% each of these restores itself when the group ends.
\newcommand{\ts@nomargin}{\ts@margfreefalse}
\AtBeginDocument{%
  \@for\ts@e:={figure,figure*,table,table*,longtable,tabular,tabularx,%
    minipage,wrapfigure,marginfigure,margintable,sidewaystable,%
    sidewaysfigure,subfigure,lstlisting,algorithm}\do{%
    \@ifundefined{\ts@e}{}{\expandafter\AtBeginEnvironment\expandafter
      {\ts@e}{\ts@nomargin}}%
  }%
  \@ifundefined{caption}{}{\pretocmd{\caption}{\ts@nomargin}{}{}}%
  \@ifundefined{footnote}{}{\pretocmd{\footnote}{\ts@nomargin}{}{}}%
  \@ifundefined{marginnote}{}{\pretocmd{\marginnote}{\ts@nomargin}{}{}}%
  % The lists in the front matter re-execute whatever a caption contained, so
  % a cited caption would put its reference in the margin of the list of
  % figures. These three are not environments and do not have to be grouped,
  % so the flag is put back explicitly rather than left to a group to restore.
  \@for\ts@e:={tableofcontents,listoffigures,listoftables}\do{%
    \@ifundefined{\ts@e}{}{%
      \expandafter\pretocmd\csname\ts@e\endcsname{\ts@nomargin}{}{}%
      \expandafter\apptocmd\csname\ts@e\endcsname{\ts@margfreetrue}{}{}%
    }%
  }%
  % Swap the three in last, so that the copies taken here are biblatex's own
  % and not something another package has already wrapped.
  \let\ts@save@cite\cite
  \let\ts@save@parencite\parencite
  \let\ts@save@textcite\textcite
  \let\cite\ts@cite
  \let\parencite\ts@parencite
  \let\textcite\ts@textcite
}

\makeatother

% ---------------------------------------------------------------------------
% 10. The front matter lists
%
% Every entry in the table of contents and in the two lists of floats is a
% hyperlink, so hyperref's linkcolor coloured all of them and the pages read
% as solid teal. The colour is worth keeping where it marks structure and not
% where it marks every line, so the lists are set with black links and the
% part and chapter entries are coloured back in.
%
% Two things have to be arranged for that. A link sets its own colour inside
% the entry, so as long as the entry text is a link nothing outside it can
% recolour one level and not another. `linktoc=page' makes only the page number
% a link, which leaves the text an ordinary piece of type. KOMA's font hooks
% for the part and chapter entries then reach it.
%
% Patching \l@part and \l@chapter was tried first and does not work here: the
% lists run under \etocstandardlines, and etoc typesets them with copies of
% those macros that it took when it was loaded, long before this file runs.
% ---------------------------------------------------------------------------

\makeatletter

\addtokomafont{partentry}{\color{ThesisPrimary}}
\addtokomafont{chapterentry}{\color{ThesisPrimary}}

% A caption is written to the list of floats whole, and the captions of the
% imported papers run to several sentences, so the two lists became walls of
% text. Shortening them at the source is not available: a caption belongs to
% its paper and those are never edited, and by the time the entry reaches the
% list it is wrapped in the label prefixing of thesis_import.sty and in
% hyperref's link, so it cannot be taken apart by pattern either.
%
% It can be cut after it is typeset. The entry goes into a box, and if the box
% is wider than the space a single line has, the overflow is clipped away and
% an ellipsis marks that something was dropped. This never inspects the
% caption, so any macro inside it is safe.
\RequirePackage{trimclip}

\newlength{\ts@entrywidth}
\newsavebox{\ts@entrybox}

\newcommand{\ts@oneline}[1]{%
    % Room for one line: the measure, less the indent this level uses, less
    % the margin the page number needs, less room for the ellipsis.
    \setlength{\ts@entrywidth}{\dimexpr\linewidth-1.5em-\@tocrmarg-2em\relax}%
    \sbox\ts@entrybox{#1}%
    \ifdim\wd\ts@entrybox>\ts@entrywidth
        \clipbox{0pt 0pt {\dimexpr\wd\ts@entrybox-\ts@entrywidth\relax} 0pt}%
                {\usebox\ts@entrybox}%
        \textcolor{ThesisGrey}{\,\dots}%
    \else
        \usebox\ts@entrybox
    \fi}

% \@dottedtocline is not used for these two, because a cut caption ends in an
% ellipsis and a row of leader dots immediately after it hides the ellipsis, so
% the cut reads as a fault rather than as a decision. The page number is set
% flush right with plain space in front of it instead. Otherwise this follows
% \@dottedtocline: \@tempdima has to be set before the entry is typeset,
% because the \numberline inside it sizes the number box from it.
\newcommand*{\ts@floatentry}[2]{%
    \ifnum \c@tocdepth >\z@
        \vskip \z@ \@plus.2\p@
        {\leftskip 1.5em \rightskip \@tocrmarg \parindent \z@
         \parfillskip -\rightskip
         \interlinepenalty\@M
         \leavevmode
         \@tempdima 2.3em
         \advance\leftskip \@tempdima \hskip -\leftskip
         \ts@oneline{#1}\nobreak\hfill\nobreak
         \hb@xt@\@pnumwidth{\hss #2}%
         \par}%
    \fi}
\renewcommand*{\l@figure}{\ts@floatentry}
\renewcommand*{\l@table}{\ts@floatentry}

% One call for the whole block, so that main.tex carries no styling. The
% group keeps the black links and the taller measure local to the lists.

    \endgroup}

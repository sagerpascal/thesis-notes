% Packages by Pascal
% \l@algorithm uses '@' which is not a letter outside \makeatletter.
% Wrap the redefinition so it parses correctly and clear the conflict
% with algorithm2e (loaded earlier by kao) before loading the algorithm
% package.
\makeatletter
\let\listofalgorithms\relax
\let\l@algorithm\relax
\makeatother
\usepackage{algorithm}
% kao.sty loads algorithm2e, which claims \For, \While, \If and the rest of the
% pseudocode vocabulary. algpseudocode then cannot define them, because
% \newcommand refuses a name that exists, and it fails silently: the imported
% papers end up with algorithm2e's \For and algpseudocode's \EndFor in the same
% algorithm, which does not compile. Clearing the names first lets algpseudocode
% win, which is the syntax every imported paper is written in. Nothing in the
% thesis uses algorithm2e's own \begin{algorithm} syntax.
\makeatletter
\@for\ip@a:={For,EndFor,ForAll,EndForAll,ForEach,While,EndWhile,Repeat,%
Until,If,EndIf,Else,ElseIf,ElsIf,Switch,Case,Do,Return,State,Require,Ensure,%
Comment,Function,EndFunction,Procedure,EndProcedure,Loop,EndLoop}\do{%
  \expandafter\let\csname\ip@a\endcsname\relax}
\makeatother
